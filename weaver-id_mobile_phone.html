function verifyNFC(uid) { showState('loading'); fetch(`${BACKEND_API_URL}?nfc_uid=${encodeURIComponent(uid)}`) .then(response => response.json().then(data => ({ status: response.status, body: data }))) .then(res => { if (res.body.status === 'success') { document.getElementById('weaver-id').textContent = '#' + res.body.data.id; document.getElementById('weaver-name').textContent = res.body.data.name; document.getElementById('weaver-business').textContent = res.body.data.business; showState('success'); } else { showError(res.body.message || 'Verification failed.'); } }) .catch(() => { showError('Unable to connect to verification server.'); }); } function showError(msg) { document.getElementById('error-message').textContent = msg; showState('error'); } function showState(state) { document.getElementById('state-loading').classList.add('d-none'); document.getElementById('state-success').classList.add('d-none'); document.getElementById('state-error').classList.add('d-none'); document.getElementById('state-input').classList.add('d-none'); document.getElementById('rescan-btn').classList.add('d-none'); document.getElementById(`state-${state}`).classList.remove('d-none'); if (state === 'success' || state === 'error') { document.getElementById('rescan-btn').classList.remove('d-none'); } } function resetView() { window.history.pushState({}, document.title, window.location.pathname); document.getElementById('manual-nfc-input').value = ''; showState('input'); }
